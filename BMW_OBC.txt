  lgfx::Panel_ST7789 _panel_instance;
  lgfx::Bus_SPI _bus_instance;

  LGFX(void) {
    {
      auto cfg = _bus_instance.config();
      cfg.spi_host = SPI2_HOST;
      cfg.freq_write = 40000000;
      cfg.pin_sclk = PIN_SCLK;
      cfg.pin_mosi = PIN_MOSI;
      cfg.pin_miso = PIN_MISO;

      cfg.pin_cs = PIN_CS;

      cfg.readable = false;
      cfg.invert = false;
      cfg.rgb_order = false;
      cfg.bus_shared = true;
      _panel_instance.config(cfg);
    }
    setPanel(&_panel_instance);
  }
};

void my_disp_flush(lv_disp_drv_t* disp, const lv_area_t* area, lv_color_t* color_p) {
  uint32_t w = (area->x2 - area->x1 + 1);
  uint32_t h = (area->y2 - area->y1 + 1);
  tft.startWrite();
  tft.setAddrWindow(area->x1, area->y1, w, h);
  tft.writePixels((lgfx::rgb565_t*)&color_p->full, w * h);
  tft.endWrite();
  lv_disp_flush_ready(disp);
}

// === SOUND DIRECT ===
void sendMP3Command(byte command, byte dat1, byte dat2) {
  byte buffer[10] = { 0x7E, 0xFF, 0x06, command, 0x00, dat1, dat2, 0x00, 0x00, 0xEF };
  uint16_t sum = 0;
  for (int i = 1; i < 7; i++) sum += buffer[i];
  sum = -sum;
  buffer[7] = (byte)(sum >> 8);
  buffer[8] = (byte)(sum);
  myMP3.write(buffer, 10);
}

void playTrack(int folder, int track) {
  if (millis() - last_sound_time > 2000) {  // Anti-spam
    sendMP3Command(0x0F, (byte)folder, (byte)track);
    last_sound_time = millis();
  }
}

// === MOVED UP: checkDailyReset (BEFORE update_gui) ===
void checkDailyReset(int current_day) {
  if (last_saved_day == -1) {
    last_saved_day = current_day;
    return;
  }
  if (current_day != last_saved_day) {
    resetDaily();
    last_saved_day = current_day;
  }
}

void update_gui() {
  lv_label_set_text_fmt(label_speed, "%d", speed_kmh);
  lv_arc_set_value(arc_rpm, rpm);
  lv_label_set_text_fmt(label_sat_count, "SAT: %d", sat_count);

  // Main Panel Volt/Temp
  String vMain = String(volt, 1) + " V";
  lv_label_set_text(label_main_volt, vMain.c_str());

  if (tempC > -50 && tempC < 150) {
    lv_label_set_text_fmt(label_main_temp, "%d C", (int)tempC);
  } else {
    lv_label_set_text(label_main_temp, "-- C");
  }

  // RPM Colors
  if (rpm < 3000) lv_obj_set_style_arc_color(arc_rpm, lv_color_hex(0x00A0FF), LV_PART_INDICATOR);
  else if (rpm < 5500) lv_obj_set_style_arc_color(arc_rpm, lv_color_hex(0x00FF00), LV_PART_INDICATOR);
  else if (rpm < 6500) lv_obj_set_style_arc_color(arc_rpm, lv_color_hex(0xFF9000), LV_PART_INDICATOR);
  else lv_obj_set_style_arc_color(arc_rpm, lv_color_hex(0xFF0000), LV_PART_INDICATOR);

  // Temp Arc
  if (tempC < -50 || tempC > 150) {
    lv_label_set_text(label_temp_val, "--");
    lv_arc_set_value(arc_temp, 0);
    lv_obj_set_style_arc_color(arc_temp, lv_color_hex(0x202020), LV_PART_INDICATOR);
  } else {
    lv_arc_set_value(arc_temp, (int)tempC);
    lv_label_set_text_fmt(label_temp_val, "%d", (int)tempC);
    if (tempC < 50) lv_obj_set_style_arc_color(arc_temp, lv_color_hex(0x00A0FF), LV_PART_INDICATOR);
    else if (tempC > 105) lv_obj_set_style_arc_color(arc_temp, lv_color_hex(0xFF0000), LV_PART_INDICATOR);
    else lv_obj_set_style_arc_color(arc_temp, lv_color_hex(0x00FF00), LV_PART_INDICATOR);

    // Update MAX Temp
    if (tempC > stat_temp_max) stat_temp_max = tempC;
    String tMaxStr = "MAX: " + String((int)stat_temp_max) + " C";
    lv_label_set_text(label_temp_max, tMaxStr.c_str());
  }

  // Volt Arc
  String vVal = String(volt, 1);
  lv_label_set_text(label_volt_val, vVal.c_str());

  int arc_val = (int)(volt * 10);
  if (arc_val < 100) arc_val = 100;
  if (arc_val > 160) arc_val = 160;
  lv_arc_set_value(arc_volt, arc_val);

  if (volt < 11.8 || volt > 15.0) lv_obj_set_style_arc_color(arc_volt, lv_color_hex(0xFF0000), LV_PART_INDICATOR);
  else lv_obj_set_style_arc_color(arc_volt, lv_color_hex(0x00A0FF), LV_PART_INDICATOR);

  if (volt > 5.0) {
    if (stat_volt_min > volt) stat_volt_min = volt;
    if (stat_volt_max < volt) stat_volt_max = volt;

    String vMin = "MIN: " + String(stat_volt_min, 1);
    String vMax = "MAX: " + String(stat_volt_max, 1);
    lv_label_set_text(label_volt_min, vMin.c_str());
    lv_label_set_text(label_volt_max, vMax.c_str());
  }

  // Service
  unsigned long hours = service_seconds / 3600;
  unsigned long km = service_meters / 1000;

  if (hours >= SERVICE_LIMIT_HOURS || km >= SERVICE_LIMIT_KM) service_alert_active = true;
  else service_alert_active = false;

  if (service_alert_active) {
    static unsigned long blink_timer = 0;
    if (millis() - blink_timer > 500) {
      blink_timer = millis();
      blink_state = !blink_state;
      if (blink_state) {
        lv_label_set_text(label_service_status, "SERVICE REQUIRED!");
        lv_obj_set_style_text_color(label_service_status, lv_color_hex(0xFF0000), 0);
      } else {
        lv_label_set_text(label_service_status, "");
      }
    }
  } else {
    lv_label_set_text(label_service_status, "SYSTEM OK");
    lv_obj_set_style_text_color(label_service_status, lv_color_hex(0x00FF00), 0);
  }

  int remaining_h = SERVICE_LIMIT_HOURS - hours;
  if (remaining_h < 0) remaining_h = 0;
  int greens_active = map(remaining_h, 0, SERVICE_LIMIT_HOURS, 0, 5);
  for (int i = 0; i < 5; i++) {
    if (i < greens_active) lv_obj_set_style_bg_color(bar_service[i], lv_color_hex(0x00FF00), 0);
    else lv_obj_set_style_bg_color(bar_service[i], lv_color_hex(0x222222), 0);
  }

  lv_label_set_text_fmt(label_service_hours, "Intv: %lu h / %d", hours, SERVICE_LIMIT_HOURS);
  lv_label_set_text_fmt(label_service_km, "Intv: %lu km / %d", km, SERVICE_LIMIT_KM);
  lv_label_set_text_fmt(label_total_engine, "TOTAL ENGINE: %lu h", engine_total_seconds / 3600);

  float trip = daily_meters / 1000.0;
  String tripStr = "DAILY: " + String(trip, 1) + " km";
  lv_label_set_text(label_daily_trip, tripStr.c_str());

  DateTime now = rtc.now();
  lv_label_set_text_fmt(label_clock_time, "%02d:%02d", now.hour(), now.minute());
  lv_label_set_text_fmt(label_clock_date, "%02d / %02d / %d", now.day(), now.month(), now.year());

  checkDailyReset(now.day());
  if (weatherUpdated) {
    String tStr = String(tempAir, 0) + " C";
    lv_label_set_text(label_weather_big, tStr.c_str());
    lv_label_set_text(label_weather_data, weatherDesc.c_str());
  }
}

void checkAlarms() {
  if (tempC > 102) {
    if (!alarm_temp_triggered) {
      playTrack(1, 2);
      alarm_temp_triggered = true;
    }
  } else {
    if (tempC < 100) alarm_temp_triggered = false;
  }
  if (volt > 1.0 && (volt < 11.5 || volt > 15.0)) {
    if (!alarm_volt_triggered) {
      playTrack(1, 3);
      alarm_volt_triggered = true;
    }
  } else {
    if (volt > 11.8 && volt < 14.8) alarm_volt_triggered = false;
  }
}

void updateWeather() {
  if (WiFi.status() != WL_CONNECTED) return;
  WiFiClient client;
  HTTPClient http;
  String url = "http://api.openweathermap.org/data/2.5/weather?lat=" + String(LATITUDE) + "&lon=" + String(LONGITUDE) + "&appid=" + OWM_KEY + "&units=metric&lang=en";
  http.begin(client, url);
  if (http.GET() == 200) {
    String payload = http.getString();
    JsonDocument doc;
    deserializeJson(doc, payload);
    tempAir = doc["main"]["temp"];
    const char* d = doc["weather"][0]["main"];
    weatherDesc = String(d);
    weatherUpdated = true;
  }
  http.end();
}

void performOTA() {
  tft.fillScreen(TFT_BLACK);
  tft.setTextColor(0xFD20, TFT_BLACK);
  tft.setTextSize(2);
  tft.setCursor(10, 50);
  tft.println("UPDATING...");
  WiFiClientSecure client;
  client.setInsecure();
  HTTPClient http;
  http.setTimeout(30000);
  http.begin(client, FIRMWARE_URL);
  if (http.GET() == 200) {
    int len = http.getSize();
    if (Update.begin(len)) {
      WiFiClient* stream = http.getStreamPtr();
      uint8_t buff[128];
      int total = 0;
      while (http.connected() && (total < len)) {
        esp_task_wdt_reset();
        size_t size = stream->available();
        if (size) {
          int c = stream->readBytes(buff, ((size > sizeof(buff)) ? sizeof(buff) : size));
          Update.write(buff, c);
          total += c;
          tft.fillRect(10, 100, map(total, 0, len, 0, 300), 20, 0x07E0);
        }
      }
      if (Update.end() && Update.isFinished()) {
        tft.fillScreen(0x07E0);
        tft.setCursor(50, 100);
        tft.println("SUCCESS!");
        delay(1000);
        ESP.restart();
      }
    }
  }
  http.end();
  tft.fillScreen(0xF800);
  delay(2000);
}

void loop() {
  lv_timer_handler();
  ArduinoOTA.handle();
  server.handleClient();
  esp_task_wdt_reset();
  if (target_tab_index != -1) {
    if (target_tab_index >= 0 && target_tab_index <= 5) {
      switchTabSafe(target_tab_index);
      playTrack(1, 4);
    }
    target_tab_index = -1;
  }

  static unsigned long last_update = 0;
  if (millis() - last_update > 100) {
    unsigned long dt = millis() - last_update;
    last_update = millis();

    if (demo_mode) {
      static int demo_rpm_dir = 1;
      rpm += (200 * demo_rpm_dir);
      if (rpm > 6500) demo_rpm_dir = -1;
      if (rpm < 800) demo_rpm_dir = 1;
      speed_kmh = rpm / 30;
      if (tempC < 90) tempC += 0.5;
      volt = 13.8 + ((float)random(0, 5) / 10.0);
    } else {
      int max_gps_read = 100;
      while (gpsSerial.available() > 0 && max_gps_read-- > 0) gps.encode(gpsSerial.read());

      noInterrupts();
      int pr = rpm_pulses;
      rpm_pulses = 0;
      interrupts();
      rpm = ((pr * 60 * 1000) / (int)dt / 3) + corr_rpm;
      if (rpm < 0) rpm = 0;
      if (gps.speed.isValid()) {
        speed_kmh = gps.speed.kmph();
        float mps = gps.speed.mps();
        if (mps > 1.0) {
          float dist = (mps * (dt / 1000.0));
          service_meters += dist;
          daily_meters += dist;
        }
      } else {
        speed_kmh = 0;
      }
      sat_count = gps.satellites.value();

      float adc = 0;
      for (int i = 0; i < 5; i++) adc += analogRead(PIN_VOLT);
      volt = ((adc / 5.0) / 4095.0) * 3.3 * 5.6 + corr_volt;

      if (millis() - last_temp_request > 1000) {
        if (temp_requested) {
          float t = sensors.getTempCByIndex(0);
          if (t > -50 && t < 150) tempC = t + corr_temp;
          else tempC = -127;
          temp_requested = false;
        }
        sensors.requestTemperatures();
        temp_requested = true;
        last_temp_request = millis();
      }

      static unsigned long sec_tick = 0;
      if (millis() - sec_tick > 1000) {
        sec_tick = millis();
        if (rpm > 300) {
          engine_total_seconds++;
          service_seconds++;
        }
        if (engine_total_seconds % 60 == 0) {
          pref.putUInt("sec", engine_total_seconds);
          pref.putUInt("srv_s", service_seconds);
          pref.putUInt("srv_m", service_meters);
          pref.putUInt("day_m", daily_meters);
        }
      }
      checkAlarms();
    }

    static unsigned long weather_timer = 0;
    if (millis() - weather_timer > 600000 || (!weatherUpdated && millis() > 10000 && millis() - weather_timer > 30000)) {
      weather_timer = millis();
      updateWeather();
    }
    update_gui();
  }

  int btn = digitalRead(PIN_BUTTON);
  int current_tab = lv_tabview_get_tab_act(tv);
  if (btn == LOW) {
    if (btn_press_start == 0) btn_press_start = millis();
    if (!btn_long_press_handled && current_tab == 3 && (millis() - btn_press_start > 12000)) {
      resetService();
      btn_long_press_handled = true;
    }
  } else {
    if (btn_press_start != 0 && !btn_long_press_handled) {
      if (millis() - btn_press_start > 50) {
        int act = lv_tabview_get_tab_act(tv);
        act++;
        if (act > 5) act = 0;
        switchTabSafe(act);
        playTrack(1, 4);
      }
    }
    btn_press_start = 0;
    btn_long_press_handled = false;
  }
  delay(5);
}
